cmake_minimum_required(VERSION 3.13)

set(PICO_BOARD pico2 CACHE STRING "")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "")
if(NOT PICO_SDK_PATH AND NOT DEFINED ENV{PICO_SDK_PATH})
  include(FetchContent)
  FetchContent_Populate(pico_sdk QUIET
    GIT_REPOSITORY https://github.com/raspberrypi/pico-sdk
    GIT_TAG 98a542c1a62fb549ffb5d66a3e5892b06276b670
    GIT_SUBMODULES lib/tinyusb
    SOURCE_DIR ${CMAKE_BINARY_DIR}/_deps/pico_sdk-src
    BINARY_DIR ${CMAKE_BINARY_DIR}/_deps/pico_sdk-build
    SUBBUILD_DIR ${CMAKE_BINARY_DIR}/_deps/pico_sdk-subbuild)
  set(PICO_SDK_PATH ${CMAKE_BINARY_DIR}/_deps/pico_sdk-src)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_definitions(PICO_NO_BI_PROGRAM_BUILD_DATE=1)
include(pico_sdk_import.cmake)
project(xombe_stick VERSION 0.1.0 LANGUAGES C CXX ASM)

set(XOMBE_STICK_DESCRIBE "" CACHE STRING "git describe output to use instead of running git")
set(describe ${XOMBE_STICK_DESCRIBE})
if(NOT describe)
  execute_process(COMMAND git describe --tags --match "v*" --always --dirty WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    OUTPUT_VARIABLE describe OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(EXISTS ${CMAKE_SOURCE_DIR}/.git/index)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS .git/HEAD .git/index)
  endif()
endif()
set(XOMBE_STICK_VERSION_STRING "xombe-stick ${PROJECT_VERSION}")
if(describe AND NOT describe STREQUAL "v${PROJECT_VERSION}")
  string(APPEND XOMBE_STICK_VERSION_STRING " (${describe})")
endif()
message(STATUS "Firmware version: ${XOMBE_STICK_VERSION_STRING}")
pico_sdk_init()

include(FetchContent)
set(XOMBE_GIT_REPOSITORY https://github.com/x0rro/xombe CACHE STRING "xombe's git repository")
set(XOMBE_GIT_TAG 2f0a8e744645b51c1749a39c8a4ee581fef1f20a CACHE STRING "xombe commit to build")
set(xombe_repo ${XOMBE_GIT_REPOSITORY})
if(NOT IS_ABSOLUTE "${xombe_repo}" AND NOT xombe_repo MATCHES "^[^/]*:")
  get_filename_component(xombe_repo "${xombe_repo}" ABSOLUTE BASE_DIR ${CMAKE_CURRENT_SOURCE_DIR})
endif()
FetchContent_Declare(xombe GIT_REPOSITORY ${xombe_repo} GIT_TAG ${XOMBE_GIT_TAG})
set(XOMBE_KERNELS q15 CACHE STRING "codec kernels, float or q15 (the firmware needs q15 for real time)")
FetchContent_MakeAvailable(xombe)

target_compile_options(xombe PRIVATE -O3 -Wdouble-promotion -fstack-usage)
set(XOMBE_EXTRA_FLAGS "" CACHE STRING "extra compile flags for xombe, e.g. -ffp-contract=off")
separate_arguments(XOMBE_EXTRA_LIST UNIX_COMMAND "${XOMBE_EXTRA_FLAGS}")
target_compile_options(xombe PRIVATE ${XOMBE_EXTRA_LIST})
option(XOMBE_LTO "link-time optimization of xombe into the firmware" ON)
if(XOMBE_LTO)
  target_compile_options(xombe PRIVATE -flto)
  target_link_options(xombe INTERFACE -flto -O3)
endif()

set(DV3K_FTDI_PRODUCT_3000 USB-3000 CACHE STRING "the FTDI 3000 firmware's USB product string, which D2XX hosts match")
set(DV3K_FTDI_PRODUCT_3003 USB-3003 CACHE STRING "the FTDI 3003 firmware's USB product string, which D2XX hosts match")
foreach(model 3000 3003)
  foreach(usb cdc ftdi)
    if(model EQUAL 3000)
      set(name xombe_stick_dv3k)
    else()
      set(name xombe_stick_dv3003)
    endif()
    if(usb STREQUAL ftdi)
      set(name ${name}_ftdi)
    endif()
    add_executable(${name} src/dv3k/dv3k.c src/dv3k/usb_${usb}.c src/dv3k/chip.c
      src/dv3k/xombe_backend.c)
    target_compile_definitions(${name} PRIVATE DV3K_MODEL=${model} DV3K_VERSION="${XOMBE_STICK_VERSION_STRING}")
    if(usb STREQUAL ftdi)
      target_compile_definitions(${name} PRIVATE DV3K_FTDI DV3K_FTDI_PRODUCT="${DV3K_FTDI_PRODUCT_${model}}")
    endif()
    target_include_directories(${name} PRIVATE src/dv3k)
    target_compile_options(${name} PRIVATE -Wall -Wextra)
    target_link_libraries(${name} xombe::xombe pico_stdlib pico_multicore pico_unique_id hardware_clocks
      hardware_vreg tinyusb_device)
    pico_enable_stdio_uart(${name} 0)
    pico_set_binary_type(${name} copy_to_ram)
    pico_set_program_version(${name} "${XOMBE_STICK_VERSION_STRING}")
    pico_add_extra_outputs(${name})
    list(APPEND firmware ${name})
  endforeach()
endforeach()
add_custom_target(firmware DEPENDS ${firmware})
